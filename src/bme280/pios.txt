# Raspberry Pi OS -- I2C-specific setup for the BME280.
# Run this BEFORE setup-common.txt.
#
# Raspberry Pi OS does NOT enable I2C by default -- it must be turned on
# explicitly (unlike Ubuntu's Raspberry Pi images, see ubuntu-24.txt).

sudo apt update
sudo apt upgrade
sudo reboot

# Enable I2C:
sudo raspi-config
# -> Interface Options -> I2C -> Yes
# Headless/scripted equivalent:
#   sudo raspi-config nonint do_i2c 0
# (sets dtparam=i2c_arm=on in /boot/firmware/config.txt on Bookworm+,
#  or /boot/config.txt on older Raspberry Pi OS releases)
sudo reboot

sudo apt install i2c-tools
sudo apt install python3-smbus
i2cdetect -y 1
# Note which address the sensor shows at -- 0x76 or 0x77.
# probe1.py must be edited to match if it's not 0x77.

# The user created by Raspberry Pi Imager is normally added to the i2c
# group automatically -- confirm rather than assume:
groups
# If "i2c" is missing from that list:
#   sudo usermod -aG i2c $USER
#   sudo reboot
